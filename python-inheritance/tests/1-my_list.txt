The ``MyList`` class behaves like a list and prints a sorted copy.

    >>> MyList = __import__('1-my_list').MyList
    >>> values = MyList([1, 4, 2, 3, 5])
    >>> values.print_sorted()
    [1, 2, 3, 4, 5]
    >>> values
    [1, 4, 2, 3, 5]
    >>> MyList().print_sorted()
    []
    >>> MyList([-1, -5, 2, 0]).print_sorted()
    [-5, -1, 0, 2]
    >>> MyList([3, 1, 3, 2]).print_sorted()
    [1, 2, 3, 3]
    >>> isinstance(MyList(), list)
    True
    >>> appended = MyList()
    >>> appended.append(3)
    >>> appended.append(1)
    >>> appended.append(2)
    >>> appended
    [3, 1, 2]
    >>> str(appended)
    '[3, 1, 2]'
    >>> result = appended.print_sorted()
    [1, 2, 3]
    >>> result is None
    True
    >>> appended
    [3, 1, 2]
    >>> ordered = MyList()
    >>> ordered.append(1)
    >>> ordered.append(2)
    >>> ordered.append(3)
    >>> ordered.print_sorted()
    [1, 2, 3]
    >>> negatives = MyList()
    >>> negatives.append(-1)
    >>> negatives.append(-5)
    >>> negatives.append(2)
    >>> negatives.print_sorted()
    [-5, -1, 2]
